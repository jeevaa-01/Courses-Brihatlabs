### Scenario

HitaVir Tech processes sales data from branches across India.

The pipeline must:

- Read raw sales and product files.
- Apply explicit schemas.
- Standardize text.
- Parse dates.
- Validate required fields.
- Remove duplicates.
- Separate valid and rejected records.
- Calculate revenue.
- Enrich sales using product master data.
- Aggregate by region and product.
- Rank products by regional revenue.
- Write Parquet outputs.
- Partition clean sales by region.
- Log the complete run.
- Include tests.

### Extract module

Create `src/extract.py`:

```python
from pathlib import Path

from pyspark.sql import DataFrame, SparkSession
from pyspark.sql.types import StructType


def read_csv(
    spark: SparkSession,
    path: Path,
    schema: StructType
) -> DataFrame:
    """Read a CSV file with an explicit schema."""
    if not path.exists():
        raise FileNotFoundError(
            f"Input file not found: {path}"
        )

    return (
        spark.read
        .option("header", True)
        .option("mode", "PERMISSIVE")
        .schema(schema)
        .csv(str(path))
    )
```

### Transform module

Create `src/transform.py`:

```python
from pyspark.sql import DataFrame
from pyspark.sql import functions as F
from pyspark.sql.window import Window


def validate_required_columns(
    df: DataFrame,
    required_columns: set[str]
) -> None:
    """Validate required DataFrame columns."""
    missing_columns = required_columns - set(df.columns)

    if missing_columns:
        raise ValueError(
            f"Missing required columns: "
            f"{sorted(missing_columns)}"
        )


def standardize_sales(df: DataFrame) -> DataFrame:
    """Standardize raw sales values."""
    return (
        df
        .withColumn(
            "transaction_id",
            F.upper(
                F.trim(F.col("transaction_id"))
            )
        )
        .withColumn(
            "transaction_date",
            F.try_to_date(
                F.col("transaction_date"),
                "yyyy-MM-dd"
            )
        )
        .withColumn(
            "region",
            F.upper(F.trim(F.col("region")))
        )
        .withColumn(
            "product",
            F.initcap(F.trim(F.col("product")))
        )
        .withColumn(
            "quantity",
            F.col("quantity").cast("integer")
        )
        .withColumn(
            "unit_price",
            F.col("unit_price").cast("double")
        )
    )


def add_validation_columns(
    df: DataFrame,
    allowed_regions: list[str]
) -> DataFrame:
    """Add record status and rejection reason."""
    return (
        df
        .withColumn(
            "rejection_reason",
            F.when(
                F.col("transaction_id").isNull(),
                F.lit("MISSING_TRANSACTION_ID")
            )
            .when(
                F.col("transaction_date").isNull(),
                F.lit("INVALID_TRANSACTION_DATE")
            )
            .when(
                ~F.col("region").isin(allowed_regions),
                F.lit("INVALID_REGION")
            )
            .when(
                F.col("product").isNull(),
                F.lit("MISSING_PRODUCT")
            )
            .when(
                F.col("quantity").isNull()
                | (F.col("quantity") <= 0),
                F.lit("INVALID_QUANTITY")
            )
            .when(
                F.col("unit_price").isNull()
                | (F.col("unit_price") < 0),
                F.lit("INVALID_UNIT_PRICE")
            )
        )
        .withColumn(
            "record_status",
            F.when(
                F.col("rejection_reason").isNull(),
                F.lit("VALID")
            ).otherwise(F.lit("REJECTED"))
        )
    )


def deduplicate_sales(df: DataFrame) -> DataFrame:
    """Keep one row per transaction ID."""
    dedup_window = (
        Window
        .partitionBy("transaction_id")
        .orderBy(
            F.col("transaction_date").asc_nulls_last()
        )
    )

    return (
        df
        .withColumn(
            "row_number",
            F.row_number().over(dedup_window)
        )
        .filter(F.col("row_number") == 1)
        .drop("row_number")
    )


def split_valid_and_rejected(
    df: DataFrame
) -> tuple[DataFrame, DataFrame]:
    """Split valid and rejected records."""
    valid_df = (
        df
        .filter(F.col("record_status") == "VALID")
        .drop("record_status", "rejection_reason")
    )

    rejected_df = df.filter(
        F.col("record_status") == "REJECTED"
    )

    return valid_df, rejected_df


def enrich_sales(
    sales_df: DataFrame,
    products_df: DataFrame
) -> DataFrame:
    """Join sales with product master and add revenue."""
    clean_products_df = (
        products_df
        .withColumn(
            "product",
            F.initcap(F.trim(F.col("product")))
        )
        .dropDuplicates(["product"])
    )

    return (
        sales_df
        .join(
            F.broadcast(clean_products_df),
            on="product",
            how="left"
        )
        .withColumn(
            "revenue",
            F.round(
                F.col("quantity")
                * F.col("unit_price"),
                2
            )
        )
        .withColumn(
            "transaction_year",
            F.year("transaction_date")
        )
        .withColumn(
            "transaction_month",
            F.month("transaction_date")
        )
    )


def build_regional_summary(
    df: DataFrame
) -> DataFrame:
    """Build regional sales summary."""
    return (
        df
        .groupBy("region")
        .agg(
            F.round(
                F.sum("revenue"),
                2
            ).alias("total_revenue"),
            F.sum(
                "quantity"
            ).alias("total_quantity"),
            F.countDistinct(
                "transaction_id"
            ).alias("transaction_count"),
            F.round(
                F.avg("unit_price"),
                2
            ).alias("average_unit_price")
        )
        .orderBy(
            F.col("total_revenue").desc()
        )
    )


def build_product_summary(
    df: DataFrame
) -> DataFrame:
    """Build ranked product summary by region."""
    summary_df = (
        df
        .groupBy(
            "region",
            "product",
            "category"
        )
        .agg(
            F.round(
                F.sum("revenue"),
                2
            ).alias("total_revenue"),
            F.sum(
                "quantity"
            ).alias("total_quantity"),
            F.countDistinct(
                "transaction_id"
            ).alias("transaction_count")
        )
    )

    rank_window = (
        Window
        .partitionBy("region")
        .orderBy(
            F.col("total_revenue").desc()
        )
    )

    return (
        summary_df
        .withColumn(
            "revenue_rank",
            F.dense_rank().over(rank_window)
        )
        .orderBy(
            "region",
            "revenue_rank",
            "product"
        )
    )
```

### Load module

Create `src/load.py`:

```python
from pathlib import Path

from pyspark.sql import DataFrame


def write_parquet(
    df: DataFrame,
    path: Path,
    partition_columns: list[str] | None = None
) -> None:
    """Write DataFrame as Parquet."""
    writer = df.write.mode("overwrite")

    if partition_columns:
        writer = writer.partitionBy(
            *partition_columns
        )

    writer.parquet(str(path))


def write_csv(
    df: DataFrame,
    path: Path
) -> None:
    """Write DataFrame as CSV."""
    (
        df.write
        .mode("overwrite")
        .option("header", True)
        .csv(str(path))
    )
```

### Pipeline

Create `src/pipeline.py`:

```python
import logging
import sys
import time

from src.config import (
    ALLOWED_REGIONS,
    CLEAN_SALES_PATH,
    LOG_PATH,
    PRODUCT_SUMMARY_PATH,
    PRODUCTS_PATH,
    RAW_SALES_PATH,
    REGIONAL_SUMMARY_PATH,
    REJECTED_PATH,
    REQUIRED_SALES_COLUMNS,
)
from src.extract import read_csv
from src.load import write_csv, write_parquet
from src.schemas import product_schema, sales_schema
from src.spark_session import create_spark_session
from src.transform import (
    add_validation_columns,
    build_product_summary,
    build_regional_summary,
    deduplicate_sales,
    enrich_sales,
    split_valid_and_rejected,
    standardize_sales,
    validate_required_columns,
)


def configure_logging() -> None:
    """Configure application logging."""
    LOG_PATH.parent.mkdir(
        parents=True,
        exist_ok=True
    )

    logging.basicConfig(
        level=logging.INFO,
        format=(
            "%(asctime)s | %(levelname)s | "
            "%(name)s | %(message)s"
        ),
        handlers=[
            logging.FileHandler(
                LOG_PATH,
                encoding="utf-8"
            ),
            logging.StreamHandler(sys.stdout),
        ],
        force=True,
    )


def run_pipeline() -> None:
    """Run HitaVir Tech PySpark pipeline."""
    configure_logging()
    logger = logging.getLogger(
        "hitavir_pyspark_pipeline"
    )

    spark = create_spark_session()
    spark.sparkContext.setLogLevel("WARN")

    started_at = time.perf_counter()

    try:
        logger.info("Pipeline started")

        raw_sales_df = read_csv(
            spark,
            RAW_SALES_PATH,
            sales_schema
        )

        products_df = read_csv(
            spark,
            PRODUCTS_PATH,
            product_schema
        )

        validate_required_columns(
            raw_sales_df,
            REQUIRED_SALES_COLUMNS
        )

        input_count = raw_sales_df.count()
        logger.info(
            "Input sales rows: %s",
            input_count
        )

        standardized_df = standardize_sales(
            raw_sales_df
        )

        validated_df = add_validation_columns(
            standardized_df,
            ALLOWED_REGIONS
        )

        deduplicated_df = deduplicate_sales(
            validated_df
        )

        valid_df, rejected_df = (
            split_valid_and_rejected(
                deduplicated_df
            )
        )

        valid_df = valid_df.cache()

        valid_count = valid_df.count()
        rejected_count = rejected_df.count()

        logger.info(
            "Valid rows: %s",
            valid_count
        )
        logger.info(
            "Rejected rows: %s",
            rejected_count
        )

        enriched_df = enrich_sales(
            valid_df,
            products_df
        )

        regional_summary_df = (
            build_regional_summary(
                enriched_df
            )
        )

        product_summary_df = (
            build_product_summary(
                enriched_df
            )
        )

        write_parquet(
            enriched_df,
            CLEAN_SALES_PATH,
            partition_columns=["region"]
        )

        write_csv(
            rejected_df,
            REJECTED_PATH
        )

        write_parquet(
            regional_summary_df,
            REGIONAL_SUMMARY_PATH
        )

        write_parquet(
            product_summary_df,
            PRODUCT_SUMMARY_PATH
        )

        valid_df.unpersist()

        elapsed_seconds = (
            time.perf_counter() - started_at
        )

        logger.info(
            "Clean output: %s",
            CLEAN_SALES_PATH
        )
        logger.info(
            "Regional summary: %s",
            REGIONAL_SUMMARY_PATH
        )
        logger.info(
            "Product summary: %s",
            PRODUCT_SUMMARY_PATH
        )
        logger.info(
            "Pipeline completed in %.3f seconds",
            elapsed_seconds
        )

    except Exception:
        logger.exception("Pipeline failed")
        raise

    finally:
        spark.stop()


if __name__ == "__main__":
    run_pipeline()
```

### Capstone rubric (peer-review style)

Review your own project — or a peer's — against these criteria before moving on:

| Criterion | Meets expectations | Needs revision |
|---|---|---|
| Correctness | Log shows 11 input → 5 valid / 5 rejected; rejection reasons match the [expected valid records](/courses/pyspark/module-5-capstone-and-testing/data-engineering-mini-project#expected-valid-records) | Counts or reasons differ from expected |
| Code structure | Config, schemas, extract, transform, load, and pipeline are separate modules with clear names | Logic concentrated in one file or unclear naming |
| Validation | Rejected records preserved with explicit `rejection_reason` values | Records silently dropped or reasons missing |
| Observability | Run logs include counts, output paths, duration, and status | Sparse or missing logging |
| Tests | `pytest -q` reports 2 passed; at least one edge case covered | Tests failing or absent |

:::info[📌 Key takeaways]

- A production-style pipeline separates configuration, extraction, transformation, loading, and orchestration.
